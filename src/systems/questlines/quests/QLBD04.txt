Item _gold_ gold range 300 to 600

--	Quest start-up:
	ql name _ysolde_ Ysolde_Marnhel
	place npc _ysolde_ at _hall_
	log 1010 step 0
	start timer _wait_

_closed_ task:
	ql act breach-closed 1
	say 1020
	log 1011 step 1
	stop timer _wait_
	start timer _meet_

_clicked_ task:
	clicked npc _ysolde_

_notyet_ task:
	when _clicked_ and not _closed_
	say 1015
	clear _clicked_ _notyet_

_paid_ task:
	when _clicked_ and _closed_
	say 1013
	give pc _gold_
	end quest

_wait_ task:
	say 1003
	end quest

_meet_ task:
	say 1003
	end quest
