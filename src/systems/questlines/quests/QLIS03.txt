Item _note_ letter used 1011

--	Quest start-up:
	place npc _master_ at _tavern_
	start timer _word_

_word_ task:
	give pc _note_ notify 1012
	log 1010 step 0
	start timer _wait_

_struck_ task:
	ql act ship-struck 1
	log 1013 step 1

_qgclicked_ task:
	clicked npc _master_

_clearclick_ task:
	when _qgclicked_ and not _struck_
	say 1014
	clear _qgclicked_ _clearclick_

_paid_ task:
	when _struck_ and _qgclicked_
	say 1015
	give pc _gold_
	end quest

_wait_ task:
	say 1003
	end quest
