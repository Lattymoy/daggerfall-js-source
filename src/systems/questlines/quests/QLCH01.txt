--	Quest start-up:
	ql name _ysolde_ Ysolde_Marnhel
	place npc _ysolde_ at _hall_
	start task _courier_

_courier_ task:
	get item _letter_
	log 1010 step 0
	start timer _wait_

_slain_ task:
	ql act elite-foe-slain 3
	say 1020
	log 1016 step 1
	stop timer _wait_
	start timer _meet_

_clicked_ task:
	clicked npc _ysolde_

_notyet_ task:
	when _clicked_ and not _slain_
	say 1015
	clear _clicked_ _notyet_

_paid_ task:
	when _clicked_ and _slain_
	say 1013
	give pc _gold_
	end quest

_wait_ task:
	say 1003
	end quest

_meet_ task:
	say 1003
	end quest
