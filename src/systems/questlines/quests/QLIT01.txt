QBN:
Item _card_ letter used 1011
Item _gold_ gold range 100 to 200

Person _regular_ group Resident1 local anyInfo 1014
Place _inn_ local tavern

Clock _courier_ 00:30 01:30
Clock _wait_ 30.00:00 0 flag 1 range 0 1

--	Quest start-up:
	place npc _regular_ at _inn_
	start timer _courier_

_courier_ task:
	give pc _card_ notify 1012
	log 1010 step 0
	start timer _wait_

_won_ task:
	ql act cards-won 1

_told_ task:
	when _courier_ and _won_
	log 1015 step 1

_clicked_ task:
	clicked npc _regular_

_early_ task:
	when _clicked_ and not _won_
	say 1016
	clear _clicked_ _early_

_paid_ task:
	when _clicked_ and _won_
	say 1013
	give pc _gold_
	end quest

_wait_ task:
	say 1003
	end quest
