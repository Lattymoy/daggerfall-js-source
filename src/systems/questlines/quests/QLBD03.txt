QBN:
Item _letter_ letter used 1011
Item _gold_ gold range 300 to 600

Person _ysolde_ faction The_Mages_Guild female anyInfo 1014

Place _hall_ remote magery

Clock _wait_ 60.00:00 0 flag 1 range 0 1
Clock _courier_ 01:00 03:00
Clock _meet_ 30.00:00 0 flag 1 range 0 1

--	Quest start-up:
	ql name _ysolde_ Ysolde_Marnhel
	log 1010 step 0
	start timer _wait_

_broken_ task:
	ql act rite-broken 1
	say 1020
	stop timer _wait_
	start timer _courier_

_courier_ task:
	place npc _ysolde_ at _hall_
	give pc _letter_ notify 1012
	log 1015 step 1
	start timer _meet_

_met_ task:
	clicked npc _ysolde_
	say 1013
	give pc _gold_
	end quest

_wait_ task:
	say 1003
	end quest

_meet_ task:
	say 1003
	end quest
