--	Quest start-up:
	pc at _glenmoril_ set _atcoven_
	start timer _courier_

variable _atcoven_

_courier_ task:
	give pc _letter_ notify 1012
	reveal _glenmoril_
	log 1010 step 0
	start timer _wait_

_arrived_ task:
	when _courier_ and _atcoven_
	start task _met_

_met_ task:
	say 1013
	ask 1030 choices 1031 _toburn_ 1032 _toguild_ 1033 _totemple_

variable _toburn_
variable _toguild_
variable _totemple_

_burnway_ task:
	when _toburn_
	say 1034
	outcome _burned_
	give pc _gold_
	end quest

_guildway_ task:
	when _toguild_
	place npc _mage_ at _hall_
	get item _spindle_ saying 1035
	log 1041 step 1

_templeway_ task:
	when _totemple_
	place npc _priest_ at _temple_
	get item _spindle_ saying 1036
	log 1042 step 1

_sold_ task:
	toting _spindle_ and _mage_ clicked
	say 1037
	outcome _guild_
	give pc _gold_
	end quest

_cut_ task:
	toting _spindle_ and _priest_ clicked
	say 1038
	outcome _temple_
	give pc _gold_
	end quest

_wait_ task:
	say 1003
	end quest
