--	Quest start-up:
	ql name _imrah_ Imrah_Bel-Tavir
	place npc _imrah_ at _tavern_
	start task _courier_

_courier_ task:
	get item _letter_
	have _letter_ set _book_
	log 1010 step 0
	start timer _wait_

_book_ task:
	get item _guide_

_boarded_ task:
	ql act ketch-boarded 1
	say 1020
	log 1016 step 1
	stop timer _wait_
	start timer _meet_

_clicked_ task:
	clicked npc _imrah_

_notyet_ task:
	when _clicked_ and not _boarded_
	say 1015
	clear _clicked_ _notyet_

_paid_ task:
	when _clicked_ and _boarded_
	say 1013
	give pc _gold_
	end quest

_wait_ task:
	say 1003
	end quest

_meet_ task:
	say 1003
	end quest
