--	Quest start-up:
	ql name _ysolde_ Ysolde_Marnhel
	place npc _ysolde_ at _hall_
	reveal _lair_
	log 1010 step 0
	give item _weirkey_ to _lieutenant_
	place foe _lieutenant_ at _lair_
	place foe _bound_ at _lair_
	pc at _lair_ set _atlair_ saying 1015
	have _weirkey_ set _haskey_
	start timer _wait_

	say 1016
	log 1011 step 1

_handed_ task:
	toting _weirkey_ and _ysolde_ clicked
	say 1020
	give pc _gold_
	end quest

	say 1003
	end quest
