QBN:
Item _gold_ gold range 100 to 200

Person _smith_ group Armorer local anyInfo 1014

--	Quest start-up:
	place npc _smith_ at _shop_
	log 1010 step 0
	start timer _wait_

_forged_ task:
	ql act piece-reforged 1
	log 1011 step 1

_clicked_ task:
	clicked npc _smith_

_early_ task:
	when _clicked_ and not _forged_
	say 1012
	clear _clicked_ _early_

_paid_ task:
	when _clicked_ and _forged_
	say 1013
	give pc _gold_
	end quest

_wait_ task:
	say 1003
	end quest
