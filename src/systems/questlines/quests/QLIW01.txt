QBN:
Item _order_ letter used 1011
Item _gold_ gold range 200 to 400

Person _captain_ factiontype Province local anyInfo 1014
Place _inn_ local tavern

Clock _courier_ 00:30 01:30
Clock _wait_ 30.00:00 0 flag 1 range 0 1

--	Quest start-up:
	place npc _captain_ at _inn_
	start timer _courier_

_courier_ task:
	give pc _order_ notify 1012
	log 1010 step 0
	start timer _wait_

_raid_ task:
	ql act raid-repelled 1

_told_ task:
	when _courier_ and _raid_
	log 1015 step 1

_clicked_ task:
	clicked npc _captain_

_early_ task:
	when _clicked_ and not _raid_
	say 1016
	clear _clicked_ _early_

_paid_ task:
	when _clicked_ and _raid_
	say 1013
	give pc _gold_
	end quest

_wait_ task:
	say 1003
	end quest
