_courier_ task:
	give pc _letter_ notify 1012
	log 1010 step 0

	stop timer _wait_
	start timer _meet_

	say 1015
	clear _clicked_ _notyet_

_paid_ task:
	when _clicked_ and _felled_
	say 1013
	give pc _gold_
	end quest

_wait_ task:
	say 1003
	end quest

	end quest
