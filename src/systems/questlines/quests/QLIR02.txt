--	Quest start-up:
	place npc _master_ at _inn_
	place npc _seller_ at _yard_
	log 1010 step 0
	start timer _wait_

_collected_ task:
	clicked npc _seller_
	say 1012
	get item _load_
	log 1011 step 1

_paid_ task:
	toting _load_ and _master_ clicked
	say 1013
	give pc _gold_
	end quest

_wait_ task:
	say 1003
	end quest
