-- hand: read it, or carry it to Ysolde shut (the Mages Guild's standing, on the sheet). Lore: the faithful's rite is the
-- port's own, as "On the Burning Doors" tells it (systems/portBooks.js).

QBN:
Item _letter_ letter used 1011
Item _ritebook_ item class 10 subclass 12
Item _gold_ gold range 300 to 600

Person _ysolde_ faction The_Mages_Guild female anyInfo 1014

Place _hall_ remote magery
Place _coven_ remote dungeon

Clock _courier_ 01:00 03:00
Clock _wait_ 60.00:00 0 flag 1 range 0 1

Foe _summoner_ is Sorcerer
Foe _faithful_ is Mage
Foe _faithful2_ is Mage

--	Quest start-up:
	ql name _ysolde_ Ysolde_Marnhel
	place npc _ysolde_ at _hall_
	start timer _courier_

_courier_ task:
	give pc _letter_ notify 1012
	log 1010 step 0
	reveal _coven_
	give item _ritebook_ to _summoner_
	place foe _summoner_ at _coven_
	place foe _faithful_ at _coven_
	place foe _faithful2_ at _coven_
	pc at _coven_ set _atcoven_ saying 1015
	have _ritebook_ set _hasbook_
	start timer _wait_

variable _atcoven_

_slain_ task:
	killed 1 _summoner_ saying 1013

_hasbook_ task:
	ask 1020 choices 1021 _readit_ 1022 _keptshut_

variable _readit_
variable _keptshut_

_reading_ task:
	when _readit_
	say 1023
	log 1024 step 1

_shut_ task:
	when _keptshut_
	log 1025 step 1

_handed_ task:
	toting _ritebook_ and _ysolde_ clicked

_paidread_ task:
	when _handed_ and _readit_
	say 1026
	outcome _read_
	give pc _gold_
	end quest

_paidshut_ task:
	when _handed_ and _keptshut_
	say 1027
	outcome _sealed_
	give pc _gold_
	end quest

_wait_ task:
	say 1003
	end quest
