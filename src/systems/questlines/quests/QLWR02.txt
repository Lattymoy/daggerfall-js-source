	ql name _corvane_ Corvane_Tull
	place npc _corvane_ at _tavern_
	log 1010 step 0
	start timer _wait_

_taken_ task:

	say 1020
	log 1016 step 1
	stop timer _wait_
	start timer _meet_

	say 1015
	clear _clicked_ _notyet_

	say 1013
	give pc _gold_
	end quest

_wait_ task:
	say 1003
	end quest

	end quest
