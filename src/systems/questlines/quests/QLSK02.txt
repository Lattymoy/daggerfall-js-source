--	Quest start-up:
	ql name _imrah_ Imrah_Bel-Tavir
	place npc _imrah_ at _tavern_
	start timer _courier_

_courier_ task:
	give pc _letter_ notify 1012
	log 1010 step 0
	start timer _wait_

_fed_ task:
	ql act ketch-cell-banked 1
	say 1020
	log 1016 step 1
	stop timer _wait_
	start timer _meet_

_clicked_ task:
	clicked npc _imrah_

_notyet_ task:
	when _clicked_ and not _fed_
	say 1015
	clear _clicked_ _notyet_

_paid_ task:
	when _clicked_ and _fed_
	say 1013
	get item _thunder_
	give pc _gold_
	end quest

_wait_ task:
	say 1003
	end quest

_meet_ task:
	say 1003
	end quest
