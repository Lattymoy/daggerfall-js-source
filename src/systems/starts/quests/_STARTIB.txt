--	Quest start-up:
	place item _commission_ at _hold_
	reveal _hold_
	pc at _daggerfall_ set _incity_
	_commission_ used do _opened_
	log 1010 step 0
	say 1020
	say 1021

_found_ task:
	clicked item _commission_
	log 1011 step 1
	remove log step 0
	say 1022

_opened_ task:
	log 1012 step 2
	remove log step 0
	remove log step 1
	reveal _daggerfall_

variable _incity_

_reported_ task:
	when _opened_ and _incity_
	start quest _BRISIEN
	end quest saying 1023
